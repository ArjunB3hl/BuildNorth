# minicpmo-utils 1.0.6 declares librosa==0.9.0, but the demo needs librosa>=0.10.2 (0.9.x crashes without
# pkg_resources on current setuptools; demo requirements.txt). The demo's own image ends up with a newer librosa
# because pip installs its requirements after minicpmo-utils. This override makes that explicit for the resolver and
# stays on the 0.10/0.11 line: librosa 1.0.0 (2026-08-11) is a new major version that minicpmo-utils was not written
# against. Verify the audio path on the GPU smoke test.
librosa>=0.10.2,<1.0
